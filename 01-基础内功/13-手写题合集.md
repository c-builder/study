## 1. 实现 new

```javascript
function myNew(Constructor, ...args) {
  const obj = Object.create(Constructor.prototype);
  const result = Constructor.apply(obj, args);
  return result instanceof Object ? result : obj;
}
```

## 2. 实现 call / apply / bind

```javascript
Function.prototype.myCall = function (ctx, ...args) {
  ctx = ctx ?? globalThis;
  const sym = Symbol('fn');
  ctx[sym] = this;
  const result = ctx[sym](...args);
  delete ctx[sym];
  return result;
};

Function.prototype.myApply = function (ctx, args = []) {
  return this.myCall(ctx, ...args);
};

Function.prototype.myBind = function (ctx, ...preset) {
  const fn = this;
  return function bound(...args) {
    // new 调用时 this 指向实例
    if (new.target) return new fn(...preset, ...args);
    return fn.apply(ctx, [...preset, ...args]);
  };
};
```

## 3. 实现 instanceof

```javascript
function myInstanceof(obj, Constructor) {
  let proto = Object.getPrototypeOf(obj);
  while (proto) {
    if (proto === Constructor.prototype) return true;
    proto = Object.getPrototypeOf(proto);
  }
  return false;
}
```

## 4. 深拷贝（循环引用 + 特殊类型）

```javascript
function deepClone(val, cache = new WeakMap()) {
  if (val === null || typeof val !== 'object') return val;
  if (cache.has(val)) return cache.get(val);
  if (val instanceof Date) return new Date(val);
  if (val instanceof RegExp) return new RegExp(val);

  const clone = Array.isArray(val) ? [] : {};
  cache.set(val, clone);
  for (const key of Reflect.ownKeys(val)) {
    clone[key] = deepClone(val[key], cache);
  }
  return clone;
}
```

## 5. 防抖 debounce

```javascript
function debounce(fn, delay, immediate = false) {
  let timer = null;
  return function (...args) {
    const callNow = immediate && !timer;
    clearTimeout(timer);
    timer = setTimeout(() => {
      timer = null;
      if (!immediate) fn.apply(this, args);
    }, delay);
    if (callNow) fn.apply(this, args);
  };
}
```

## 6. 节流 throttle

```javascript
function throttle(fn, delay) {
  let last = 0;
  return function (...args) {
    const now = Date.now();
    if (now - last >= delay) {
      last = now;
      fn.apply(this, args);
    }
  };
}

// 时间戳 + 定时器结合版（首尾都执行）
function throttleFull(fn, delay) {
  let timer = null, last = 0;
  return function (...args) {
    const now = Date.now();
    const remaining = delay - (now - last);
    clearTimeout(timer);
    if (remaining <= 0) {
      last = now;
      fn.apply(this, args);
    } else {
      timer = setTimeout(() => {
        last = Date.now();
        fn.apply(this, args);
      }, remaining);
    }
  };
}
```

## 7. 柯里化 curry

```javascript
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) return fn.apply(this, args);
    return (...more) => curried(...args, ...more);
  };
}
// curry((a, b, c) => a + b + c)(1)(2)(3) → 6
```

## 10. EventEmitter（发布订阅）

```javascript
class EventEmitter {
  constructor() { this.events = {}; }
  on(event, fn) {
    (this.events[event] ||= []).push(fn);
    return this;
  }
  off(event, fn) {
    if (!fn) { delete this.events[event]; return this; }
    this.events[event] = this.events[event]?.filter(f => f !== fn);
    return this;
  }
  emit(event, ...args) {
    this.events[event]?.forEach(fn => fn.apply(this, args));
    return this;
  }
  once(event, fn) {
    const wrapper = (...args) => { fn(...args); this.off(event, wrapper); };
    return this.on(event, wrapper);
  }
}
```

## 11. Promise A+ 简易版

```javascript
const PENDING = 'pending', FULFILLED = 'fulfilled', REJECTED = 'rejected';

class MyPromise {
  constructor(executor) {
    this.state = PENDING;
    this.value = undefined;
    this.callbacks = [];
    const resolve = (val) => {
      if (this.state !== PENDING) return;
      this.state = FULFILLED;
      this.value = val;
      this.callbacks.forEach(cb => cb.onFulfilled?.(val));
    };
    const reject = (err) => {
      if (this.state !== PENDING) return;
      this.state = REJECTED;
      this.value = err;
      this.callbacks.forEach(cb => cb.onRejected?.(err));
    };
    try { executor(resolve, reject); } catch (e) { reject(e); }
  }
  then(onFulfilled, onRejected) {
    return new MyPromise((resolve, reject) => {
      const run = (fn, val) => {
        queueMicrotask(() => {
          try {
            const x = typeof fn === 'function' ? fn(val) : val;
            x instanceof MyPromise ? x.then(resolve, reject) : resolve(x);
          } catch (e) { reject(e); }
        });
      };
      if (this.state === FULFILLED) run(onFulfilled, this.value);
      else if (this.state === REJECTED) run(onRejected, this.value);
      else this.callbacks.push({
        onFulfilled: (v) => run(onFulfilled, v),
        onRejected: (e) => run(onRejected, e),
      });
    });
  }
}
```

## 12. Promise.all / race / allSettled

```javascript
Promise.myAll = function (promises) {
  return new Promise((resolve, reject) => {
    const results = [];
    let count = 0;
    const n = promises.length;
    if (n === 0) return resolve([]);
    promises.forEach((p, i) => {
      Promise.resolve(p).then(
        val => { results[i] = val; if (++count === n) resolve(results); },
        reject
      );
    });
  });
};

Promise.myRace = function (promises) {
  return new Promise((resolve, reject) => {
    promises.forEach(p => Promise.resolve(p).then(resolve, reject));
  });
};
```

## 13. 并发控制（限流调度器）

```javascript
async function pool(tasks, limit) {
  const results = [];
  const executing = new Set();
  for (const [i, task] of tasks.entries()) {
    const p = Promise.resolve().then(() => task()).then(r => { results[i] = r; });
    executing.add(p);
    p.finally(() => executing.delete(p));
    if (executing.size >= limit) await Promise.race(executing);
  }
  await Promise.all(executing);
  return results;
}
```

```javascript
Promise.myAllSettled = (promises) =>
  Promise.all(promises.map(p =>
    Promise.resolve(p)
      .then(value => ({ status: 'fulfilled', value }))
      .catch(reason => ({ status: 'rejected', reason }))

Promise.myAny = (promises) => new Promise((resolve, reject) => {
  const errors = [];
  let rejected = 0;
  if (!promises.length) return reject(new AggregateError([], 'All rejected'));
  promises.forEach((p, i) =>
    Promise.resolve(p).then(resolve, e => {
      errors[i] = e;
      if (++rejected === promises.length) reject(new AggregateError(errors, 'All rejected'));
    })
  );
});
```

## 15. retry + 超时

```javascript
function retry(fn, { times = 3, delay = 1000 } = {}) {
  return async (...args) => {
    let lastErr;
    for (let i = 0; i < times; i++) {
      try { return await fn(...args); }
      catch (e) { lastErr = e; if (i < times - 1) await new Promise(r => setTimeout(r, delay * (i + 1))); }
    }
    throw lastErr;
  };
}

function withTimeout(promise, ms) {
  return Promise.race([
    promise,
    new Promise((_, reject) => setTimeout(() => reject(new Error('Timeout')), ms)),
  ]);
}
```

## 16. 深比较 isEqual

```javascript
function isEqual(a, b) {
  if (a === b) return true;
  if (a == null || b == null || typeof a !== 'object' || typeof b !== 'object') return false;
  if (Array.isArray(a) !== Array.isArray(b)) return false;
  const keysA = Object.keys(a), keysB = Object.keys(b);
  if (keysA.length !== keysB.length) return false;
  return keysA.every(k => keysB.includes(k) && isEqual(a[k], b[k]));
}
```

## 17. 对象扁平化 / 路径取值

```javascript
function flatten(obj, prefix = '', res = {}) {
  for (const [k, v] of Object.entries(obj)) {
    const key = prefix ? `${prefix}.${k}` : k;
    if (v && typeof v === 'object' && !Array.isArray(v)) flatten(v, key, res);
    else res[key] = v;
  }
  return res;
}

function get(obj, path, defaultVal) {
  return path.split('.').reduce((o, k) => o?.[k], obj) ?? defaultVal;
}
// get({ a: { b: 1 } }, 'a.b') → 1
```

## 19. JSONP

```javascript
function jsonp(url, params = {}) {
  return new Promise((resolve, reject) => {
    const cb = `jsonp_${Date.now()}`;
    params.callback = cb;
    const qs = new URLSearchParams(params).toString();
    const script = document.createElement('script');
    script.src = `${url}?${qs}`;
    window[cb] = (data) => { resolve(data); cleanup(); };
    script.onerror = () => { reject(new Error('JSONP failed')); cleanup(); };
    function cleanup() {
      delete window[cb];
      script.remove();
    }
    document.head.appendChild(script);
  });
}
```

## 面试高频问答（追问链）

> 大厂对一个考点通常追问到能力边界：**概念 → 机制 → 边界 → ⭐ 原理（触底）→ 实战（落地）**。实战层是区分「背过」和「做过」的关键。手写题会追问边界、复杂度与规范细节。

### 链一：call/bind/this

1. **概念**：bind 和 call 区别？→ bind 返回新函数不立即执行。
2. **机制**：箭头函数能 bind 吗？→ 不能改 this，无 own this。
3. **边界**：bind 后再 new 会怎样？→ new 优先，this 指向新实例（用 `new.target` 判断）。
4. ⭐ **原理（触底）**：手写 call 怎么处理基本类型 this、Symbol 防污染、严格模式？→ 装箱基本类型、用 Symbol 临时键避免覆盖、null 时按规范指 globalThis/undefined。
5. **实战（落地）**：白板写 bind 要注意什么？→ 处理 new.target 分支；箭头函数直接说不能 bind；写完测 `fn.bind(obj,1)(2)` 和 `new boundFn()` 两种场景。

### 链二：深拷贝与相等

1. **概念**：深拷贝怎么处理循环引用？→ WeakMap 缓存已拷贝对象。
2. **机制**：为什么用 WeakMap 不用 Map？→ 弱引用键，不阻止 GC。
3. **边界**：深拷贝还要处理哪些类型？→ Date/RegExp/Map/Set/Symbol 键/函数。
4. ⭐ **原理（触底）**：structuredClone 和手写深拷贝差异？→ 原生支持循环引用与多数内置类型，但不能拷函数/DOM/原型链，会抛错。
5. **实战（落地）**：深拷贝白板怎么写不丢分？→ 先写 WeakMap 防循环；逐类型分支 Date/RegExp/Map；主动说函数一般跳过；写完报 O(n) 并测循环引用对象。

### 链三：防抖节流与 Promise

1. **概念**：防抖、节流场景？→ 搜索输入防抖、滚动/resize 节流。
2. **机制**：leading/trailing 怎么控制？→ immediate 参数 + 时间戳/定时器组合。
3. **边界**：手写 Promise 为什么用 queueMicrotask？→ 保证 then 异步、符合 A+、避免乱序。
4. **应用**：allSettled vs all？→ all 一败全败；allSettled 全完成返回各自状态。
5. ⭐ **原理（触底）**：retry 为什么要指数退避 + 抖动（jitter）？→ 避免故障期集中重试压垮服务端，jitter 防止「重试风暴」同时打到。
6. **实战（落地）**：手写 Promise 面试怎么演示？→ 先写三态 + then 链；强调 queueMicrotask 异步；跑同步 resolve 和嵌套 then 两个用例；主动说符合 A+ 的要点即可，不必背全规范。
